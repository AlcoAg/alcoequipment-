    // Filter by Category Function
    function filterCategory(cat) {
      currentFilter = cat;
      const buttons = document.querySelectorAll('.cat-btn');
      buttons.forEach(btn => {
        if (btn.dataset.cat === cat) {
          btn.className = "cat-btn active-cat px-4 py-2 rounded-md font-heading text-sm uppercase font-bold tracking-wider transition bg-brand-yellow text-black";
        } else {
          btn.className = "cat-btn px-4 py-2 rounded-md font-heading text-sm uppercase font-semibold tracking-wider transition bg-brand-surface text-zinc-300 hover:text-white border border-brand-border";
        }
      });

      applyFilters();
    }

    // Filter + Search Logic
    function applyFilters() {
      const searchTerm = document.getElementById('inventorySearch').value.toLowerCase().trim();
      
      const filtered = equipmentData.filter(item => {
        const matchesCategory = (currentFilter === 'all') || (item.category === currentFilter);
        const matchesSearch = !searchTerm || 
          item.title.toLowerCase().includes(searchTerm) ||
          item.tag.toLowerCase().includes(searchTerm) ||
          (item.specs.hp && item.specs.hp.toLowerCase().includes(searchTerm)) ||
          item.description.toLowerCase().includes(searchTerm);
        
        return matchesCategory && matchesSearch;
      });

      renderInventory(filtered);
    }

    // Search Input Listener
    document.getElementById('inventorySearch').addEventListener('input', applyFilters);

    // Preset Financing Calculator from an Equipment Card
    function presetCalculator(price, name) {
      const priceSlider = document.getElementById('calcPrice');
      priceSlider.value = price;
      updateCalculator();

      const selectInterest = document.getElementById('leadInterest');
      for (let i = 0; i < selectInterest.options.length; i++) {
        if (selectInterest.options[i].text.includes(name.split(' ')[0])) {
          selectInterest.selectedIndex = i;
          break;
        }
      }

      document.getElementById('financing').scrollIntoView({ behavior: 'smooth' });
    }

    // Loan / Payment Calculator Formula
    function updateCalculator() {
      const price = parseFloat(document.getElementById('calcPrice').value);
      const down = parseFloat(document.getElementById('calcDown').value);
      const apr = parseFloat(document.getElementById('calcRate').value);
      const term = parseInt(document.getElementById('calcTerm').value, 10);

      document.getElementById('priceValDisplay').textContent = '$' + price.toLocaleString();
      document.getElementById('downValDisplay').textContent = '$' + down.toLocaleString();
      document.getElementById('rateValDisplay').textContent = apr.toFixed(1) + '%';

      const principal = Math.max(0, price - down);
      
      if (principal <= 0) {
        document.getElementById('monthlyPaymentDisplay').textContent = '$0';
        return;
      }

      const monthlyRate = (apr / 100) / 12;
      let monthlyPayment = 0;

      if (monthlyRate === 0) {
        monthlyPayment = principal / term;
      } else {
        monthlyPayment = (principal * monthlyRate * Math.pow(1 + monthlyRate, term)) / (Math.pow(1 + monthlyRate, term) - 1);
      }

      document.getElementById('monthlyPaymentDisplay').textContent = '$' + Math.round(monthlyPayment).toLocaleString();
    }

    // Calculator event listeners

      document.getElementById(id).addEventListener('input', updateCalculator);
